```dart
  /// Every topic's lessons in one round trip, each list in authored order.
  /// A topic with no lessons still gets a key, so a caller never has to tell
  /// "no lessons" apart from "topic I did not ask about".
  Future<Map<TopicId, List<Lesson>>> lessonsByTopics(
    List<TopicId> topicIds,
  ) async {
    final grouped = {for (final id in topicIds) id: <Lesson>[]};
    if (topicIds.isEmpty) return grouped;

    final rows = await (_db.select(_db.lessons)
          ..where((l) => l.topicId.isIn(topicIds))
          ..orderBy([
            (l) => OrderingTerm.asc(l.ordering),
            (l) => OrderingTerm.asc(l.id),
          ]))
        .get();

    for (final row in rows) {
      grouped[row.topicId]?.add(_hydrateLesson(row));
    }
    return grouped;
  }
```

```dart
import 'package:cse_cet_reviewer/core/storage/domain/ids.dart';

/// One lesson as a row on the map.
class StudyLesson {
  const StudyLesson({
    required this.id,
    required this.title,
    required this.estimatedMinutes,
    required this.isRead,
    required this.isBookmarked,
  });

  final LessonId id;
  final String title;
  final int estimatedMinutes;
  final bool isRead;
  final bool isBookmarked;
}

/// One topic row, and the lessons it reveals when it opens.
class StudyTopic {
  const StudyTopic({
    required this.id,
    required this.name,
    required this.lessons,
    required this.readCount,
    required this.hasBookmark,
  });

  final TopicId id;
  final String name;
  final List<StudyLesson> lessons;
  final int readCount;
  final bool hasBookmark;

  int get totalCount => lessons.length;

  /// A topic with no lessons is not finished, it is unauthored. Treating it as
  /// complete would put a board tick against content nobody has written.
  bool get isComplete => totalCount > 0 && readCount == totalCount;
}

/// One subject heading and its topics.
class StudySubject {
  const StudySubject({
    required this.id,
    required this.name,
    required this.topics,
  });

  final SubjectId id;
  final String name;
  final List<StudyTopic> topics;

  int get readCount =>
      topics.fold(0, (running, topic) => running + topic.readCount);

  int get totalCount =>
      topics.fold(0, (running, topic) => running + topic.totalCount);
}

/// The whole screen, computed.
class StudyMap {
  const StudyMap({
    required this.examShortName,
    required this.subjects,
    required this.readCount,
    required this.totalCount,
    required this.minutesLeftToRead,
  });

  final String examShortName;
  final List<StudySubject> subjects;
  final int readCount;
  final int totalCount;

  /// The sum of authored `estimatedMinutes` over unread lessons. This is a
  /// property of the material and never of the student: nothing records how
  /// long anyone read.
  final int minutesLeftToRead;

  /// Read lessons over total, rounded. Zero when the exam has no lessons,
  /// rather than a division by zero.
  int get coveredPercent =>
      totalCount == 0 ? 0 : (readCount / totalCount * 100).round();

  bool get isEmpty => totalCount == 0;
}
```

```dart
import 'package:cse_cet_reviewer/core/storage/domain/exam.dart';
import 'package:cse_cet_reviewer/core/storage/domain/ids.dart';
import 'package:cse_cet_reviewer/core/storage/domain/lesson.dart';
import 'package:cse_cet_reviewer/core/storage/domain/topic.dart';
import 'package:cse_cet_reviewer/features/lesson/domain/models/study_map.dart';

export 'package:cse_cet_reviewer/features/lesson/domain/models/study_map.dart';

/// Turns the exam's tree and the read set into everything the map draws.
///
/// The exam is the authority on which subjects exist and which topics belong
/// to them: a topic in [lessonsByTopic] that no subject lists is ignored
/// rather than counted, so a stale row in the database cannot move a figure.
StudyMap buildStudyMap({
  required Exam exam,
  required Map<SubjectId, List<Topic>> topicsBySubject,
  required Map<TopicId, List<Lesson>> lessonsByTopic,
  required Set<LessonId> readLessonIds,
  required Set<LessonId> bookmarkedLessonIds,
}) {
  final subjects = <StudySubject>[];
  var readCount = 0;
  var totalCount = 0;
  var minutesLeft = 0;

  for (final subject in exam.subjects) {
    final topics = <StudyTopic>[];

    for (final topic in topicsBySubject[subject.id] ?? const <Topic>[]) {
      final lessons = lessonsByTopic[topic.id] ?? const <Lesson>[];
      var topicRead = 0;
      var hasBookmark = false;
      final rows = <StudyLesson>[];

      for (final lesson in lessons) {
        final isRead = readLessonIds.contains(lesson.id);
        final isBookmarked = bookmarkedLessonIds.contains(lesson.id);
        if (isRead) {
          topicRead++;
        } else {
          minutesLeft += lesson.estimatedMinutes;
        }
        hasBookmark = hasBookmark || isBookmarked;

        rows.add(
          StudyLesson(
            id: lesson.id,
            title: lesson.title,
            estimatedMinutes: lesson.estimatedMinutes,
            isRead: isRead,
            isBookmarked: isBookmarked,
          ),
        );
      }

      readCount += topicRead;
      totalCount += rows.length;

      topics.add(
        StudyTopic(
          id: topic.id,
          name: topic.name,
          lessons: rows,
          readCount: topicRead,
          hasBookmark: hasBookmark,
        ),
      );
    }

    subjects.add(
      StudySubject(id: subject.id, name: subject.name, topics: topics),
    );
  }

  return StudyMap(
    examShortName: exam.shortName,
    subjects: subjects,
    readCount: readCount,
    totalCount: totalCount,
    minutesLeftToRead: minutesLeft,
  );
}
```

```dart
/// Null when no exam is chosen. The screen draws its own fallback for that,
/// so it is a state rather than an error.
final studyMapProvider = FutureProvider<StudyMap?>((ref) async {
  final focus = await ref.watch(activeExamFocusProvider.future);
  final exam = focus.exam;
  if (exam == null) return null;

  final content = ref.watch(contentRepositoryProvider);
  final topicsBySubject = <SubjectId, List<Topic>>{};
  for (final subject in exam.subjects) {
    topicsBySubject[subject.id] = await content.topicsBySubject(subject.id);
  }

  final topicIds = [
    for (final topics in topicsBySubject.values)
      for (final topic in topics) topic.id,
  ];

  return buildStudyMap(
    exam: exam,
    topicsBySubject: topicsBySubject,
    lessonsByTopic: await content.lessonsByTopics(topicIds),
    readLessonIds: await ref.watch(lessonProgressRepositoryProvider).readLessonIds(),
    bookmarkedLessonIds:
        await ref.watch(lessonBookmarksRepositoryProvider).markedIds(),
  );
});

/// Which topic is expanded, at most one, remembered per exam across visits to
/// the tab. One at a time is what bounds the page: fully expanded, UPCAT is
/// 164 rows.
class OpenTopicController extends AsyncNotifier<TopicId?> {
  @override
  Future<TopicId?> build() async {
    final examId = await _examId();
    if (examId == null) return null;
    return ref.read(appPreferencesRepositoryProvider).openTopicId(examId);
  }

  Future<void> toggle(TopicId topicId) async {
    final next = state.valueOrNull == topicId ? null : topicId;
    await _write(next);
  }

  Future<void> close() => _write(null);

  Future<void> _write(TopicId? topicId) async {
    final examId = await _examId();
    if (examId == null) return;
    await ref
        .read(appPreferencesRepositoryProvider)
        .setOpenTopicId(examId: examId, topicId: topicId);
    state = AsyncData(topicId);
  }

  Future<ExamId?> _examId() async {
    final focus = await ref.read(activeExamFocusProvider.future);
    return focus.exam?.id;
  }
}

final openTopicProvider =
    AsyncNotifierProvider<OpenTopicController, TopicId?>(
  OpenTopicController.new,
);

/// The lesson the "Last opened" row offers to resume. Null hides the row.
final lastOpenedLessonProvider = FutureProvider<Lesson?>((ref) async {
  final focus = await ref.watch(activeExamFocusProvider.future);
  final examId = focus.exam?.id;
  if (examId == null) return null;

  final lessonId = await ref
      .watch(appPreferencesRepositoryProvider)
      .lastOpenedLessonId(examId);
  if (lessonId == null) return null;

  return ref.watch(contentRepositoryProvider).lessonById(lessonId);
});
```

cache every time. It calls the repository directly instead. The provider from Task 1 exists
for callers with a stable list; if nothing uses it by the end of Plan 5, delete it.
